\titulo{Jerarquía de von Neumann}

\problema[-3mm]{}{Definimos, de manera recursiva:
    $$V_0=\varnothing,\quad 
    V_{\alpha+1}=\mathcal{P}(V_\alpha),$$ $$
    V_\gamma=\displaystyle\bigcup\{V_\beta:\beta<\gamma\}$$ para $\alpha$ un ordinal y $\gamma$ un ordinal límite.

\vspace{5mm}
\noindent Probar que 
    $
        \displaystyle\bigcup\{V_\alpha:\alpha\in\mathrm{ON}\}
    $
    es la clase universal.}
